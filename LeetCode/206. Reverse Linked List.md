```csharp
public ListNode ReverseList(ListNode head) {
    if(head is null)
        return head;

    var currNode = head;

    return funcation(currNode, null);

    ListNode funcation(ListNode currNode, ListNode prevNode){
        if (currNode.next is null)
        {
            currNode.next = prevNode;
            return currNode;
        }

        var reverseHead = funcation(currNode.next, currNode);

        if(prevNode is not null)
            currNode.next = prevNode;

        if(prevNode is null)
            currNode.next = null;

        return reverseHead;
    }
}
```

```csharp
public ListNode ReverseList(ListNode head) {
    if(head is null)
        return head;

    var currNode = head;
    ListNode prevNode = null;

    while(currNode is not null)
    {
        var nextNode = currNode.next;
        currNode.next = prevNode;
        prevNode = currNode;
        currNode = nextNode;
    }

    return prevNode;
}
```
