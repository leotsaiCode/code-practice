> BFS, use queue to traverse level by level, snapshot queue.Count as the node count of current level, depth + 1 after each level
> Time O(n) Space O(w), w = max width of tree, worst case O(n)

```csharp
public int MaxDepth(TreeNode root) {
    if(root == null)
        return 0;

    var maxDepth = 0;
    var queue = new Queue<TreeNode>();
    queue.Enqueue(root);

    while(queue.Count > 0)
    {
        var levelNodeCount = queue.Count;

        for(int i = 0;i< levelNodeCount; i++ )
        {
            var currNode = queue.Dequeue();

            if(currNode.left != null)
                queue.Enqueue(currNode.left);
            if(currNode.right != null)
                queue.Enqueue(currNode.right);
        }

        maxDepth += 1;
    }

    return maxDepth;
}
```

> DFS, pass current depth down (top-down), leaf returns its depth, parent returns the larger one of left and right
> Time O(n) Space O(h), h = height of tree, balanced tree O(log n), worst case (skewed tree) O(n)

```csharp
public int MaxDepth(TreeNode root) {
    if(root == null)
        return 0;

    return DFSReturnMaxDepth(root, 1);

    int DFSReturnMaxDepth(TreeNode currNode, int currMaxDepth)
    {
        var rightMaxDepth = 0;
        var leftMaxDepth = 0;
        if(currNode.right is not null)
            rightMaxDepth = DFSReturnMaxDepth(currNode.right, currMaxDepth + 1);

        if(currNode.left is not null)
            leftMaxDepth = DFSReturnMaxDepth(currNode.left, currMaxDepth + 1);

        if (currNode.left == null && currNode.right == null)
            return currMaxDepth;

        return rightMaxDepth > leftMaxDepth ? rightMaxDepth : leftMaxDepth;
    }
}
```
