> anagrams share the same letter counts, so use the 26-slot count as the dictionary key
> char[] instead of int[] so it can turn into a string key directly via new string(count)
> char can hold counts up to 65535, enough for strs[i].length <= 100
> Time O(n * k) Space O(n * k), n = strs.Length, k = max string length

```csharp
public IList<IList<string>> GroupAnagrams(string[] strs) {
    var groupList = new Dictionary<string,List<string>>();

    foreach(var str in strs)
    {
        var count = new char[26];

        foreach(var charr in str)
            count[charr - 'a']++;

        var key = new string(count);

        if(!groupList.TryGetValue(key, out List<string> list))
        {
            list = new List<string>();
            groupList[key] = list;
        }
        list.Add(str);
    }
    return new List<IList<string>>(groupList.Values);
}
```
