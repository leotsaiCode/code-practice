public class Solution {
    public int LengthOfLongestSubstring(string s) {

        var charLastShowIndexDict = new Dictionary<char, int>();

        var maxLen = 0;
        var leftIndex = 0;

        for(int rightIndex = 0;rightIndex<s.Length; rightIndex++){
            var rightChar = s[rightIndex];
            if(charLastShowIndexDict.TryGetValue(rightChar, out var dictIndex)
                //"ccbbcc"
                && dictIndex >= leftIndex)
            {
                // make sure no repeat char  my ex:"abcarcwrtyrhj"
                leftIndex = dictIndex +1;
            }

            maxLen = maxLen > (rightIndex-leftIndex+1) ? maxLen :(rightIndex-leftIndex+1);

            charLastShowIndexDict[rightChar] = rightIndex;
        }

        return maxLen;
    }
}
