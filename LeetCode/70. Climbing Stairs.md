> brute force recursion: from currentStep, try taking 1 step or 2 steps
> reach targetStairs -> 1 way, overshoot -> 0 way
> the same currentStep is recomputed many times, so the recursion tree grows like Fibonacci
> Time O(2^n) Space O(n) (call stack), Time Limit Exceeded for larger n

```csharp
// BF ver 1
public int ClimbStairs(int n) {

    return CountWays(0, n);

    int CountWays(int currentStep, int targetStairs)
    {
        var currWay = 0;
        if(currentStep == targetStairs)
            return 1;

        if(currentStep > targetStairs )
            return 0;

        var result = 0;

        result = CountWays(currentStep + 2, targetStairs);
        currWay+=result;

        result = CountWays(currentStep + 1, targetStairs);
        currWay+= result;

        return currWay;
    }
}
// BF ver 2
public int ClimbStairs(int n) {

    return CountWays(0, n);

    int CountWays(int currentStep, int targetStairs)
    {
        if (currentStep == targetStairs)
            return 1;

        if (currentStep > targetStairs)
            return 0;

        return CountWays(currentStep + 1, targetStairs)
             + CountWays(currentStep + 2, targetStairs);
    }
}
```

> memoization: cache ways from each currentStep to the end, the same currentStep reached by another path reuses the cached value
> distinct states = currentStep from 0 to n -> n states, each computed once with O(1) work
> Time O(n) Space O(n)

```csharp
public int ClimbStairs(int n) {
        var startNodeWaysDict = new Dictionary<int, int>();

        return CountWays(0, n, startNodeWaysDict);

        int CountWays(int currentStep, int targetStairs, Dictionary<int, int> startNodeWaysDict)
        {
            if(currentStep == targetStairs)
                return 1;

            if(currentStep > targetStairs )
                return 0;

            if(startNodeWaysDict.TryGetValue(currentStep, out var ways))
                return ways;

            var resultWays = CountWays(currentStep + 2, targetStairs,startNodeWaysDict)
                + CountWays(currentStep + 1, targetStairs,startNodeWaysDict);

            startNodeWaysDict[currentStep]=resultWays;

            return resultWays;
        }
    }
```

> bottom-up DP + space optimization: ways(i) = ways(i + 1) + ways(i + 2), same as Fibonacci
> every route from step i starts with +1 or +2 and the two groups never overlap, so the two counts can be added
> only the latest two results are needed, so keep two variables instead of a dictionary
> Time O(n) Space O(1)

```csharp
public int ClimbStairs(int n) {

        int nextStepWays = 1;
        int twoStepsAheadWays = 0;

        for (int step = n - 1; step >= 0; step--)
        {
            (nextStepWays, twoStepsAheadWays) =
                (nextStepWays + twoStepsAheadWays, nextStepWays);
        }

        return nextStepWays;
    }
```

```csharp
public int ClimbStairs(int n) {

    var previous = 1; // ways to reach step 0
    var current = 1;  // ways to reach step 1

    for (var step = 2; step <= n; step++)
    {
        (previous, current) = (current, previous + current);
    }

    return current;
}
```
