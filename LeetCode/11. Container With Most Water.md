> the water rule: the area is determined by the shorter of the two lines
> as the width shrinks, the area can only exceed the previous one if we find a line taller than the current shorter side

> two pointers: start with left = 0, right = length - 1 (widest container)
> area = (right - left) * min(leftHeight, rightHeight), keep the max
> move the pointer at the shorter line inward, because width shrinks every step and only a taller short side can increase the area
> equal heights -> move both

> Time O(n) Space O(1)

```csharp
    public int MaxArea(int[] height) {
        var maxArea = 0;
        var length = height.Length;
        var left = 0;
        var right = length - 1;

        while(left < right)
        {
            var rightHeight = height[right];
            var leftHeight = height[left];

            var currentArea = (right - left) * Math.Min(rightHeight, leftHeight);

            maxArea = maxArea > currentArea ? maxArea: currentArea;

            if(rightHeight > leftHeight)
            {
                left += 1;
                continue;
            }

            if(rightHeight < leftHeight)
            {
                right -= 1;
                continue;
            }

            if(leftHeight == rightHeight)
            {
                right -= 1;
                left += 1;
            }
        }

        return maxArea;
    }
```

```csharp
public int MaxArea(int[] height)
{
    var maxArea = 0;
    var left = 0;
    var right = height.Length - 1;

    while (left < right)
    {
        var width = right - left;
        var minHeight = Math.Min(height[left], height[right]);
        var currentArea = width * minHeight;

        maxArea = Math.Max(maxArea, currentArea);

        if (height[left] <= height[right])
            left++;
        else
            right--;
    }

    return maxArea;
}
```
